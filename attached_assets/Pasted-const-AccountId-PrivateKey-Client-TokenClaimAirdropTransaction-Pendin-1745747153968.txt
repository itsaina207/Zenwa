const {
    AccountId,
    PrivateKey,
    Client,
    TokenClaimAirdropTransaction,
    PendingAirdropId
} = require("@hashgraph/sdk"); // v2.46.0

async function claimAirdrop() {
  let client;
  try {
    // 1. Configurer ton compte (remplacer par tes vraies valeurs)
    const MY_ACCOUNT_ID = AccountId.fromString("0.0.5876043");
    const MY_PRIVATE_KEY = PrivateKey.fromStringECDSA("84891e1414cd4f0fa75abaf0f4f13bc46a6082234682d4a5fe88c2885a46270a");

    // 2. ID du pending airdrop que tu veux claim (à remplacer par le tien)
    const pendingAirdropId = PendingAirdropId.fromString("0.0.xxxx"); // Remplacer "xxxx"

    // 3. Private key du compte qui fait le claim
    const privateKeyReceiverAccount = PrivateKey.fromStringECDSA("xxx"); // Remplacer

    // 4. Connecter au testnet
    client = Client.forTestnet();
    client.setOperator(MY_ACCOUNT_ID, MY_PRIVATE_KEY);

    // 5. Créer la transaction de claim
    const txTokenClaimAirdrop = await new TokenClaimAirdropTransaction()
      .addPendingAirdropId(pendingAirdropId)
      .freezeWith(client);

    // 6. Signer avec la clé du compte qui réclame
    const signedTx = await txTokenClaimAirdrop.sign(privateKeyReceiverAccount);

    // 7. Envoyer la transaction
    const response = await signedTx.execute(client);

    // 8. Attendre le reçu pour savoir si c'est bien passé
    const receipt = await response.getReceipt(client);

    // 9. Afficher les résultats
    console.log("--------------------------------- Claim Token Airdrop ---------------------------------");
    console.log("Receipt status           :", receipt.status.toString());
    console.log("Transaction ID           :", response.transactionId.toString());
    console.log("Hashscan URL             :", `https://hashscan.io/testnet/tx/${response.transactionId.toString()}`);
    
  } catch (error) {
    console.error("Error during claim airdrop:", error);
  } finally {
    if (client) client.close();
  }
}

// Lancer la fonction
claimAirdrop();
